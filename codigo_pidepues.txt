package co.edu.udea.compumovil.labs20262_gr08.pidepues.data

import co.edu.udea.compumovil.labs20262_gr08.pidepues.model.Review
import com.google.firebase.auth.FirebaseAuth
import com.google.firebase.firestore.FirebaseFirestore
import kotlinx.coroutines.tasks.await
import kotlin.math.roundToInt

class ReviewRepository(
    private val db: FirebaseFirestore = FirebaseFirestore.getInstance(),
    private val auth: FirebaseAuth = FirebaseAuth.getInstance()
) {
    /**
     * Guarda la reseÃ±a y recalcula el promedio del restaurante de forma atÃ³mica.
     */
    suspend fun submitReview(
        orderId: String,
        orderShortCode: String,
        restaurantId: String,
        restaurantName: String,
        stars: Int,
        comment: String
    ): Result<Unit> = runCatching {
        require(stars in 1..5) { "La calificaciÃ³n debe estar entre 1 y 5" }

        val user = auth.currentUser ?: error("Usuario no autenticado")
        val reviewRef = db.collection("reviews").document()
        val restaurantRef = db.collection("restaurants").document(restaurantId)
        val orderRef = db.collection("orders").document(orderId)

        val review = Review(
            id = reviewRef.id,
            orderId = orderId,
            orderShortCode = orderShortCode,
            restaurantId = restaurantId,
            restaurantName = restaurantName,
            userId = user.uid,
            userName = user.displayName ?: "Cliente",
            rating = stars,
            comment = comment.trim(),
            createdAtMillis = System.currentTimeMillis()
        )

        db.runTransaction { tx ->
            val restSnap = tx.get(restaurantRef)
            val currentRating = restSnap.getDouble("rating") ?: 0.0
            val currentCount = (restSnap.getLong("reviewCount") ?: 0L).toInt()

            val newCount = currentCount + 1
            val rawAvg = ((currentRating * currentCount) + stars) / newCount
            // Redondeo a 1 decimal (ej. 4.8)
            val newAverage = (rawAvg * 10.0).roundToInt() / 10.0

            tx.set(reviewRef, review)
            tx.update(
                restaurantRef,
                mapOf(
                    "rating" to newAverage,
                    "reviewCount" to newCount
                )
            )
            tx.update(
                orderRef,
                mapOf(
                    "rated" to true,
                    "userRating" to stars
                )
            )
        }.await()
    }
}

import co.edu.udea.compumovil.labs20262_gr08.pidepues.model.Restaurant
import co.edu.udea.compumovil.labs20262_gr08.pidepues.model.RestaurantFilters

/**
 * logica de cercanÃ­a + calificaciÃ³n
 * Kotlin puro para  probarla con pruebas unitarias
 */
object RestaurantRanking {

    /**distancia a partir de la cual la cercania ya no suma puntos al ranking */
    private const val MAX_REFERENCE_KM = 10.0
    private const val WEIGHT_RATING = 0.6
    private const val WEIGHT_PROXIMITY = 0.4

    /**
     * Puntaje de Recomendados: 60 % calificaciÃ³n + 40 % cercanÃ­a (0 a 1).
     * Es la linea base que en la semana 10 la IA reemplaza/enriquece con preferencias e historial.
     */
    fun recommendationScore(rating: Float, distanceKm: Double): Double {
        val ratingPart = (rating / 5f).toDouble().coerceIn(0.0, 1.0)
        val proximityPart = 1.0 - (distanceKm / MAX_REFERENCE_KM).coerceIn(0.0, 1.0)
        return WEIGHT_RATING * ratingPart + WEIGHT_PROXIMITY * proximityPart
    }

    /**calcula la distancia de cada restaurante al usuario */
    fun withDistance(restaurants: List<Restaurant>, location: UserLocation): List<RestaurantWithDistance> =
        restaurants.map {
            RestaurantWithDistance(
                it,
                haversineKm(location.latitude, location.longitude, it.latitude, it.longitude)
            )
        }

    /**aplica los filtros (distancia, calificaciÃ³n, tipo de cocina) y el orden elegido */
    fun filterAndSort(
        items: List<RestaurantWithDistance>,
        filters: RestaurantFilters
    ): List<RestaurantWithDistance> {
        val filtered = items.filter {
            it.distanceKm <= filters.maxDistanceKm &&
                it.restaurant.rating >= filters.minRating &&
                (filters.categories.isEmpty() || it.restaurant.category in filters.categories)
        }
        val byOption: Comparator<RestaurantWithDistance> = when (filters.sort) {
            SortOption.RECOMMENDED -> compareByDescending {
                recommendationScore(it.restaurant.rating, it.distanceKm)
            }
            SortOption.NEAREST -> compareBy { it.distanceKm }
            SortOption.BEST_RATED -> compareByDescending<RestaurantWithDistance> { it.restaurant.rating }
                .thenBy { it.distanceKm }
            SortOption.FASTEST -> compareBy<RestaurantWithDistance> { it.restaurant.deliveryMinutes }
                .thenBy { it.distanceKm }
        }
        // Los restaurantes cerrados siempre van al final de la lista.
        return filtered.sortedWith(compareByDescending<RestaurantWithDistance> { it.restaurant.isOpen }.then(byOption))
    }

    /**busqueda por texto en nombre, cocina, etiqueta, categorÃ­a y palabras clave */
    fun matchesQuery(restaurant: Restaurant, query: String): Boolean {
        val tokens = query.normalized().split(" ").filter { it.isNotBlank() }
        if (tokens.isEmpty()) return true
        val haystack = listOf(
            restaurant.name, restaurant.cuisine, restaurant.dishTag,
            restaurant.category.label, restaurant.keywords
        ).joinToString(" ").normalized()
        return tokens.all { haystack.contains(it) }
    }

    fun search(items: List<RestaurantWithDistance>, query: String): List<RestaurantWithDistance> =
        items.filter { matchesQuery(it.restaurant, query) }

    private fun String.normalized(): String =
        Normalizer.normalize(this.lowercase(), Normalizer.Form.NFD)
            .replace(Regex("\\p{M}+"), "")
}

/**
 * Obtiene la ubicaciÃ³n del dispositivo con el LocationManager del sistema
 * (sin dependencias extra). Si algo falla (sin permiso, GPS apagado, tiempo agotado)
 * devuelve null y la app usa [DEMO_CENTER].
 *
 * Mejora futura (semana 11): migrar a FusedLocationProviderClient (Google Play Services).
 */
object LocationProvider {

    /** UbicaciÃ³n por defecto: El Poblado, MedellÃ­n (donde estÃ¡n los datos de ejemplo). */
    val DEMO_CENTER = UserLocation(latitude = 6.2088, longitude = -75.5671, isFromDevice = false)

    /**
     * Mientras los restaurantes sean datos de ejemplo de MedellÃ­n, una ubicaciÃ³n real
     * a mÃ¡s de este radio se ignora (si no, la lista saldrÃ­a vacÃ­a en otra ciudad).
     */
    const val DEMO_RADIUS_KM = 60.0

    private const val TIMEOUT_MS = 8_000L

    fun hasPermission(context: Context): Boolean =
        ContextCompat.checkSelfPermission(context, Manifest.permission.ACCESS_FINE_LOCATION) ==
            PackageManager.PERMISSION_GRANTED ||
            ContextCompat.checkSelfPermission(context, Manifest.permission.ACCESS_COARSE_LOCATION) ==
            PackageManager.PERMISSION_GRANTED

    @SuppressLint("MissingPermission")
    fun getCurrentLocation(context: Context, onResult: (UserLocation?) -> Unit) {
        if (!hasPermission(context)) {
            onResult(null)
            return
        }
        val manager = context.getSystemService(Context.LOCATION_SERVICE) as? LocationManager
        if (manager == null) {
            onResult(null)
            return
        }

        val providers = runCatching { manager.getProviders(true) }.getOrDefault(emptyList())

        // 1) Ãšltima ubicaciÃ³n conocida (instantÃ¡nea)
        val last = providers
            .mapNotNull { runCatching { manager.getLastKnownLocation(it) }.getOrNull() }
            .maxByOrNull { it.time }
        if (last != null) {
            onResult(last.toUserLocationOrNull())
            return
        }

        // 2) Si no hay, pedimos una sola actualizaciÃ³n (con tiempo mÃ¡ximo)
        val provider = when {
            LocationManager.NETWORK_PROVIDER in providers -> LocationManager.NETWORK_PROVIDER
            LocationManager.GPS_PROVIDER in providers -> LocationManager.GPS_PROVIDER
            else -> null
        }
        if (provider == null) {
            onResult(null)
            return
        }
        SingleUpdate(manager, onResult).start(provider)
    }

    private fun Location.toUserLocationOrNull(): UserLocation? {
        val km = haversineKm(latitude, longitude, DEMO_CENTER.latitude, DEMO_CENTER.longitude)
        return if (km <= DEMO_RADIUS_KM) UserLocation(latitude, longitude, isFromDevice = true) else null
    }

    private class SingleUpdate(
        private val manager: LocationManager,
        private val onResult: (UserLocation?) -> Unit
    ) : LocationListener {

        private val handler = Handler(Looper.getMainLooper())
        private var done = false

        @SuppressLint("MissingPermission")
        @Suppress("DEPRECATION")
        fun start(provider: String) {
            try {
                manager.requestSingleUpdate(provider, this, Looper.getMainLooper())
                handler.postDelayed({ finish(null) }, TIMEOUT_MS)
            } catch (e: SecurityException) {
                finish(null)
            }
        }

        private fun finish(result: UserLocation?) {
            if (done) return
            done = true
            handler.removeCallbacksAndMessages(null)
            runCatching { manager.removeUpdates(this) }
            onResult(result)
        }

        override fun onLocationChanged(location: Location) {
            finish(location.toUserLocationOrNull())
        }

        // En Android 9 o menor estos mÃ©todos son abstractos: hay que implementarlos.
        @Deprecated("Deprecated in Java")
        override fun onStatusChanged(provider: String?, status: Int, extras: Bundle?) = Unit
        override fun onProviderEnabled(provider: String) = Unit
        override fun onProviderDisabled(provider: String) = Unit
    }
}

            row.rb.isChecked = selected

        _binding = null

        query: String = current.query,
        filters: RestaurantFilters = current.filters,
        recents: List<String> = current.recentSearches
    ) {
        _state.value = buildState(location, query, filters, recents)
    }

        LocationProvider.getCurrentLocation(context.applicationContext) { location ->
            if (location != null) update(location = location)
        }
    }

        if (query == current.query) return
        update(query = query)
    }

    fun submitSearch(query: String) {
        val text = query.trim()
        if (text.isEmpty()) return
        update(query = text, recents = pushRecent(text))
    }

    fun selectRecent(query: String) {
        update(query = query, recents = pushRecent(query))
    }

    fun clearRecents() = update(recents = emptyList())

    private fun pushRecent(text: String): List<String> =
        (listOf(text) + current.recentSearches.filterNot { it.equals(text, ignoreCase = true) })
            .take(MAX_RECENTS)

    fun setSort(sort: SortOption) {
        if (sort == current.filters.sort) return

    }

    }

    fun clearAll() = update(query = "", filters = RestaurantFilters())

fun haversineKm(lat1: Double, lon1: Double, lat2: Double, lon2: Double): Double {
    val dLat = Math.toRadians(lat2 - lat1)
    val dLon = Math.toRadians(lon2 - lon1)
    val a = sin(dLat / 2).pow(2) +
            cos(Math.toRadians(lat1)) * cos(Math.toRadians(lat2)) * sin(dLon / 2).pow(2)
    return 2 * EARTH_RADIUS_KM * asin(sqrt(a))
}

import android.view.View
import androidx.core.view.ViewCompat
import androidx.core.view.WindowInsetsCompat

/*sumaajusta espacio de padding, margins y demas para la vista*/
fun View.applySystemBarsInsets(top: Boolean = true, bottom: Boolean = true) {
    val initialLeft = paddingLeft
    val initialTop = paddingTop
    val initialRight = paddingRight
    val initialBottom = paddingBottom

    ViewCompat.setOnApplyWindowInsetsListener(this) { view, insets ->
        val bars = insets.getInsets(
            WindowInsetsCompat.Type.systemBars() or
                WindowInsetsCompat.Type.displayCutout() or
                WindowInsetsCompat.Type.ime()
        )
        view.setPadding(
            initialLeft + bars.left,
            initialTop + if (top) bars.top else 0,
            initialRight + bars.right,
            initialBottom + if (bottom) bars.bottom else 0
        )
        insets
    }
}
